```swift
func sendVoiceCommand(_ text: String) {
    print("[Watch→Server] sendVoiceCommand('\(text)')")

}

// ConnectivityProvider.swift 第 51-79 行
private func sendToSyncServer(text: String, isUser: Bool) {
    guard let url = URL(string: syncServerURL) else { return }
    var request = URLRequest(url: url)
    request.httpMethod = "POST"
    request.setValue("application/json", forHTTPHeaderField: "Content-Type")
    
    let payload: [String: Any] = [
        "text": text,
        "isUser": isUser,
        "source": "watch",
        "timestamp": ISO8601DateFormatter().string(from: Date())
    ]
    
    request.httpBody = try? JSONSerialization.data(withJSONObject: payload)
    URLSession.shared.dataTask(with: request).resume()
}
```

## 技術細節

### HTTP POST Payload
```json
{
  "text": "訊息內容",
  "isUser": true,           // true=用戶, false=AI
  "source": "watch",        // 來源設備
  "timestamp": "2026-02-16T15:47:47Z"
}
```

### WebSocket Broadcast Message
```json
{
  "text": "訊息內容",
  "isUser": true,
  "source": "watch",
  "timestamp": "2026-02-16T15:47:47Z",
  "broadcast_time": "2026-02-16T15:47:47.123Z"
}
```

## 問題排查

### 如果 WebSocket 連線失敗
```bash

ps aux | grep demo_sync_server

python3 -c "import socket; s = socket.socket(); s.bind(('127.0.0.1', 8080)); print('Port available')"

# 測試健康檢查
curl http://localhost:8080/health
```

```bash

  -H "Content-Type: application/json" \

# 檢查伺服器日誌
```
